// Règles Firestore à publier dans Firebase Console > Firestore Database > Règles.
// Pour attribuer un administrateur, créer manuellement users/UID avec role: "admin".
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function admin() {
      return signedIn()
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    match /users/{userId} {
      allow create: if signedIn() && request.auth.uid == userId
        && request.resource.data.role == 'membre';
      allow read: if signedIn() && (request.auth.uid == userId || admin());
      allow update: if signedIn() && request.auth.uid == userId
        && request.resource.data.role == resource.data.role;
      allow delete: if false;
    }

    match /cups/{cupId} {
      allow read: if true;
      allow create, update, delete: if admin();

      match /comments/{commentId} {
        allow read: if true;
        allow create: if signedIn() && request.resource.data.authorUid == request.auth.uid;
        allow update: if false;
        allow delete: if signedIn() && (request.auth.uid == resource.data.authorUid || admin());
      }
    }

    match /casts/{castId} {
      allow read: if true;
      allow create, update, delete: if admin();
    }

    match /players/{playerId} {
      allow read: if true;
      allow create, update: if signedIn() && request.auth.uid == playerId;
      allow delete: if signedIn() && (request.auth.uid == playerId || admin());
    }

    match /news/{newsId} {
      allow read: if true;
      allow create, update, delete: if admin();
    }
  }
}
